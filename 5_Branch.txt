//To known the available branches
git branch

//To create a new branch from a specific commit id //Get the commit id from "git log --oneline"
git branch <branch_name> <commit_id>

//To move into the branch
git checkout <branch_name>
or
git switch <branch_name>

//To delete a local branch
git branch -d <branch_name>

//To delete a remote branch
git push origin -d <branch_name>

//To push the local branch changes to remote branch //i.e., pushing the changes in current branch to a remote branch
git push -u origin <branch-name>

//To merge a branch locally
git checkout main
git pull origin main
git merge <branch_name>
git push origin main

//To merge using Pull request
git checkout <branch_name>
git push origin <branch_name>
gh pr create --title "Your PR Title" --body "Description of changes"
gh pr merge --squash --delete-branch 

//To get PR status
gh pr status
