
//Step1: Creating key-pair:
ssh-keygen -t ed25519 -C "your_email@example.com"
//Use the noreply email of your github account for efficiency
//When asked for the filename to save the key, enter ~/.ssh/<filename>
//Dont use any passphrase. Just press enter. Keys will be generated.

//Step2: Adding config file for efficient use of multiple github accounts:
 cd ~/.ssh/				
 mkdir config				
 cd config					
 gvim config

//Enter the below text in config file:
    Host <your_hostname> github.com			
        Hostname github.com
        AddKeysToAgent yes				
        PreferredAuthentications publickey		
        IdentityFile ~/.ssh/<filename>

//Step3: Adding the ssh public key in Github
	Open your gitub account			
 	Settings -> SSH and GPG Keys -> New SSH Key		
  	Add the key				
   	Click on Add SSH Key

//Step4: cloning the repo: cd to the directory where you want to clone the repo
